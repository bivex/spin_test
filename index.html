Promela Source (mutex_peterson.pml) 
bool flag[2] = false;
byte turn = 0;
byte critical_count = 0;

proctype Worker(byte id) {
    byte other = 1 - id;
    do
    :: true ->
want:   flag[id] = true;
        turn = id;
        (flag[other] == false || turn == other);

in_crit:
        critical_count++;
        /* Mutual Exclusion Check */
        assert(critical_count <= 1);
        critical_count--;
        flag[id] = false;
    od
}

/* Safety Property: Mutual exclusion holds */
ltl mutex { [] (critical_count <= 1) }

    case 2: /* STATE 2 - turn = id */
        now.turn = ((P0 *)this)->id;
        break;
    case 3: /* STATE 3 - wait condition */
        if (!((now.flag[other] == 0) || (now.turn == other)))
            continue; /* backtrack in DFS */
        break;
    case 4: /* STATE 4 - assert(critical_count <= 1) */
        spin_assert(now.critical_count <= 1, "assert fail", II, tt, t);
        break;
    }
}
